#### CourseLoadThread - 课程加载线程
```python
class CourseLoadThread(QThread):
    """课程加载线程"""
    finished = pyqtSignal(bool, str)
    progress = pyqtSignal(str)

    def __init__(self, file_path):
        super().__init__()
        self.file_path = file_path

    def run(self):
        """执行加载任务"""
        try:
            self.progress.emit("正在读取Excel文件...")
            loader = CourseDataLoader()
            success = loader.load_from_excel(self.file_path)
            
            if success:
                self.progress.emit(f"加载完成: {len(loader.get_courses())} 门课程")
                self.finished.emit(True, f"成功加载 {len(loader.get_courses())} 门课程")
            else:
                self.finished.emit(False, "加载失败，请检查文件格式")
        except Exception as e:
            self.finished.emit(False, f"加载失败: {str(e)}")
```

def update_credit_statistics(self):
    """更新学分统计显示"""
    if not self.credit_manager:
        return

    try:
        # 🔍 调试：检查当前CreditManager状态
        print("🔍 [调试] 当前CreditManager状态:")
        for category, req in self.credit_manager.requirements.items():
            if req.completed_credits > 0 or req.base_completed_credits > 0:
                print(f"   📊 {category}: completed={req.completed_credits:.1f}, base={req.base_completed_credits:.1f}")

        # 更新总体统计
        total_required = self.credit_manager.get_total_required_credits()
        total_completed = self.credit_manager.get_total_completed_credits()
        completion_rate = self.credit_manager.get_overall_completion_rate() * 100
    except Exception as e:
        print(f"❌ 获取学分统计数据失败: {e}")
        return

    self.total_credits_label.setText(
        f"总学分: {total_completed:.1f}/{total_required:.1f} ({completion_rate:.1f}%)"
    )

def add_time_slot(self):
    """添加时间段"""
    try:
        weekday = self.weekday_combo.currentIndex() + 1
        start_section = self.start_section.value()
        end_section = self.end_section.value()

        if start_section > end_section:
            QMessageBox.warning(self, "错误", "开始节次不能大于结束节次")
            return

        # 获取选中的周次
        weeks = self.get_selected_weeks()
        if not weeks:
            QMessageBox.warning(self, "错误", "请至少选择一个上课周次")
            return

        # 创建时间段
        time_slot = TimeSlot(
            weekday=weekday,
            start_section=start_section,
            end_section=end_section,
            weeks=weeks
        )

        # 将结果存储在dialog中
        self.result_time_slot = time_slot
        self.accept()  # 关闭对话框并返回Accepted

    except Exception as e:
        QMessageBox.critical(self, "错误", f"添加时间段失败: {str(e)}")
```

    for button in self.week_buttons:

    """清空所有周次选择"""
    for button in self.week_buttons:

def get_selected_weeks(self):
    """获取选中的周次列表"""
    selected_weeks = []
    for i, button in enumerate(self.week_buttons):
        if button.isChecked():
            selected_weeks.append(i + 1)  # 周次从1开始
    return selected_weeks

def _set_result_content(self):
    """设置结果内容"""
    if not self.result['success']:
        # 失败情况
        content = f"❌ 测试失败\n\n错误原因:\n{self.result['error']}\n\n"
        content += "请检查文件格式和内容是否正确。\n"
        content += "详细错误信息请查看日志文件。"
        self.result_text.setStyleSheet("color: #d32f2f;")
    else:
        # 成功情况
        added_courses = self.result['added_courses']
        stats = self.result.get('stats', {})
        
        content = f"✅ 测试成功完成\n\n"
        
        if added_courses:
            content += f"📚 成功添加 {len(added_courses)} 门课程:\n"
            for course in added_courses[:5]:  # 只显示前5门
                content += f"• {course.course.name} ({course.course.teacher})\n"
            if len(added_courses) > 5:
                content += f"• ... 还有 {len(added_courses) - 5} 门课程\n"
        else:
            content += "📚 没有找到可以添加的课程\n"
            
        # 添加统计信息
        if stats:
            success_rate = (stats.get('successfully_added', 0) / max(stats.get('missing_courses', 1), 1)) * 100
            content += f"📊 统计信息:\n"
            content += f"• 总体成功率: {success_rate:.1f}%\n"
            content += f"• 成功添加: {stats.get('successfully_added', 0)} 门\n"
            content += f"• 无法添加: {stats.get('failed_to_add', 0)} 门\n"
        
        self.result_text.setStyleSheet("color: #388e3c;")
    
    self.result_text.setPlainText(content)

```
用户点击选择文件 → QFileDialog → 创建CourseLoadThread → 异步加载 → 更新界面状态
```

### 2. 时间段添加流程

```

### 3. 排课执行流程

用户点击开始排课 → 服务层调用 → 事件订阅更新 → 结果展示 → 导出功能启用
```

### 4. 学分统计更新流程
```

## 🎨 界面样式设计
