```python
    """验证配置参数的有效性"""
    if self.min_campus_transfer_time < 0:

    if not (0.0 <= self.max_credit_overflow_ratio <= 1.0):
        return False
    if self.max_solve_time_seconds <= 0:

def is_campus_conflict_allowed(self, campus1: str, campus2: str, 
                              time_gap: int) -> bool:
    """检查校区冲突是否允许"""
    if campus1 == campus2:
        return True
    
    if self.campus_conflict_mode == CampusConflictMode.DISABLED:
        return True
    
    if self.campus_conflict_mode == CampusConflictMode.DAILY:
        return False

    if self.campus_conflict_mode == CampusConflictMode.PERIOD:
        return time_gap >= self.min_campus_transfer_time
    
    return False
```

**三阶段排课策略：**
```python
def generate_schedules(self, available_courses: List[SelectedCourse],
                      max_solutions: Optional[int] = None) -> List[ScheduleResult]:
    """生成排课方案（三阶段处理）"""
    start_time = time.time()

    if max_solutions is None:
        max_solutions = self.config.max_solutions

    print(f"开始排课：{len(available_courses)} 门可选课程")
    print(f"🎯 当前模式：{self.config.credit_constraint_mode.value.upper()} ({'必需模式' if self.config.credit_constraint_mode.value == 'required' else '优化模式'})")

    # 🔧 辅助修复：重置CreditManager的临时状态，确保多次运行时不会累积
    # 重置为用户手动设置的基础已修学分，保留用户在学分设置中的配置
    original_completed = {}
    for category, requirement in self.credit_manager.requirements.items():
        original_completed[category] = requirement.completed_credits
        # 重置为用户设置的基础已修学分，而不是0.0
        requirement.completed_credits = requirement.base_completed_credits

    try:
        # 阶段1：处理无时间线上课程
        online_courses, timed_courses = self._separate_courses(available_courses)
        selected_online_courses = self._process_online_courses(online_courses)
        print(f"阶段1完成：选择了 {len(selected_online_courses)} 门无时间线上课程")

        # 阶段2：核心排课（有时间安排的课程）
        if timed_courses:
            timed_solutions = self._schedule_timed_courses(timed_courses, max_solutions)
            print(f"阶段2完成：生成了 {len(timed_solutions)} 个有时间课程方案")
        else:
            timed_solutions = [[]]  # 空的有时间课程方案

        # 阶段3：合并结果
        results = []
        for i, timed_solution in enumerate(timed_solutions):
            combined_solution = selected_online_courses + timed_solution
            result = self._create_schedule_result(
                f"方案_{i+1}",
                combined_solution,
                time.time() - start_time,
                len(available_courses)
            )
            results.append(result)

        return results

    finally:
        # 恢复原始学分状态
        for category, original_credits in original_completed.items():
            if category in self.credit_manager.requirements:
                self.credit_manager.requirements[category].completed_credits = original_credits
```

**阶段1：线上课程处理：**
```python
def _process_online_courses(self, online_courses: List[SelectedCourse]) -> List[SelectedCourse]:
    """处理无时间安排的线上课程（阶段1）"""
    print(f"阶段1：处理 {len(online_courses)} 门无时间线上课程")

    if not online_courses:
        return []

    # 获取学分要求
    credit_requirements = self._calculate_credit_requirements()
    print(f"学分要求: {credit_requirements}")

    # 按学分需求优先级排序
    sorted_online = self._sort_courses_by_priority(online_courses)

    # 贪心选择
    selected_online = []
    for course in sorted_online:
        category = course.custom_category
        if category in credit_requirements and credit_requirements[category] > 0:
            selected_online.append(course)
            credit_requirements[category] -= course.course.credits
            print(f"   ✓ 选择线上课程: {course.course.name} ({course.course.credits}学分)")

    print(f"🎉 线上课程选择完成：{len(selected_online)} 门课程")

    # 🔧 关键修复：更新CreditManager状态，将选择的线上课程学分添加到已修学分中
    # 这样阶段2在计算学分缺口时就会考虑阶段1已选择的课程，避免重复计算
    for selected_course in selected_online:
        category = selected_course.custom_category
        credits = selected_course.course.credits
        self.credit_manager.add_completed_credits(category, credits)
        print(f"   📊 更新学分状态：{category} +{credits:.1f}学分")

    return selected_online
```

**阶段2：约束满足排课：**
```python
def _constraint_satisfaction_scheduling(self, timed_courses: List[SelectedCourse],
                                      max_solutions: int) -> List[List[SelectedCourse]]:
    """完整的约束满足排课算法"""
    print(f"开始约束满足排课：{len(timed_courses)} 门有时间课程")

    # 第1步：预筛选 - 如果课程过多且冲突严重，先进行智能筛选
    if len(timed_courses) > 15:
        print(f"课程数量较多({len(timed_courses)}门)，进行智能预筛选...")
        timed_courses = self._intelligent_course_filtering(timed_courses)
        print(f"预筛选完成：保留 {len(timed_courses)} 门优质课程")

    # 第2步：按课程编码分组，实现同一课程不同班次互斥
    course_groups = self._group_courses_by_code_from_selected(timed_courses)
    print(f"课程分组完成：{len(course_groups)} 个课程组")

    # 第3步：计算学分需求
    credit_requirements = self._calculate_credit_requirements()

    # 第4步：使用回溯算法生成满足约束的方案
    solutions = []
    self._backtrack_scheduling(
        course_groups=course_groups,
        credit_requirements=credit_requirements,
        current_solution=[],
        used_time_slots=set(),
        solutions=solutions,
        max_solutions=max_solutions
    )

    print(f"约束满足排课完成：生成 {len(solutions)} 个方案")

    # 如果没有找到解，提供详细的诊断信息
    if len(solutions) == 0:
        print("🔍 排课失败诊断：")
        print(f"  - 课程组数量: {len(course_groups)}")
        print(f"  - 学分要求: {credit_requirements}")
        print(f"  - 约束模式: {self.config.credit_constraint_mode.value}")
        print(f"  - 校区冲突模式: {self.config.campus_conflict_mode.value}")

        # 检查时间冲突
        conflict_count = self._analyze_time_conflicts(timed_courses)
        print(f"  - 时间冲突数量: {conflict_count}")

        # 检查学分分布
        self._analyze_credit_distribution(timed_courses, credit_requirements)

    return solutions
```

```python
def check_all_hard_constraints(self, selected_courses: List[SelectedCourse]) -> Tuple[bool, List[ConflictInfo]]:
    """检查所有硬约束条件"""
    conflicts = []
    
    # 1. 时间冲突检查
    time_conflicts = self.check_time_conflicts(selected_courses)
    conflicts.extend(time_conflicts)
    
    # 2. 校区冲突检查
    campus_conflicts = self.check_campus_conflicts(selected_courses)
    conflicts.extend(campus_conflicts)
    
    # 3. 学分约束检查（仅在REQUIRED模式下作为硬约束）
    if self.config.credit_constraint_mode == CreditConstraintMode.REQUIRED:
        credit_conflicts = self.check_credit_constraints(selected_courses)
        conflicts.extend(credit_conflicts)
    
    return len(conflicts) == 0, conflicts
```

**时间冲突检查：**
```python
def check_time_conflicts(self, selected_courses: List[SelectedCourse]) -> List[ConflictInfo]:
    """检查时间冲突"""
    conflicts = []
    
    for i in range(len(selected_courses)):
        for j in range(i + 1, len(selected_courses)):
            course1 = selected_courses[i]
            course2 = selected_courses[j]
            
            # 检查两门课程是否有时间冲突
            if course1.has_time_conflict_with(course2):
                conflicts.append(ConflictInfo(
                    conflict_type="time",
                    course1=course1.course,
                    course2=course2.course,
                    description=f"时间冲突：{course1.course.name} 与 {course2.course.name}",
                    severity="high"
                ))
    
    return conflicts
```

**校区冲突检查：**
```python
def check_campus_conflicts(self, selected_courses: List[SelectedCourse]) -> List[ConflictInfo]:
    """检查校区冲突"""
    conflicts = []
    
    if self.config.campus_conflict_mode == CampusConflictMode.DISABLED:
        return conflicts
    
    # 按天分组检查
    daily_courses = defaultdict(list)
    for course in selected_courses:
        if not course.is_online:  # 线上课程跳过校区冲突检查
            for time_slot in course.time_slots:
                daily_courses[time_slot.weekday].append((course, time_slot))
    
    for weekday, day_courses in daily_courses.items():
        if self.config.campus_conflict_mode == CampusConflictMode.DAILY:
            # 按天检查：同一天不能有不同校区的课程
            campuses = set()
            for course, time_slot in day_courses:
                campuses.add(course.course.campus)
            
            if len(campuses) > 1:
                # 找到冲突的课程对
                campus_courses = defaultdict(list)
                for course, time_slot in day_courses:
                    campus_courses[course.course.campus].append(course)
                
                campus_list = list(campus_courses.keys())
                for i in range(len(campus_list)):
                    for j in range(i + 1, len(campus_list)):
                        campus1, campus2 = campus_list[i], campus_list[j]
                        course1 = campus_courses[campus1][0]
                        course2 = campus_courses[campus2][0]
                        
                        conflicts.append(ConflictInfo(
                            conflict_type="campus",
                            course1=course1.course,
                            course2=course2.course,
                            description=f"校区冲突：{course1.course.name}({campus1}) 与 {course2.course.name}({campus2}) 在同一天",
                            severity="medium"
                        ))
        
        elif self.config.campus_conflict_mode == CampusConflictMode.PERIOD:
            # 按时段检查：考虑跨校区转移时间
            day_courses.sort(key=lambda x: x[1].start_section)
            
            for i in range(len(day_courses) - 1):
                course1, slot1 = day_courses[i]
                course2, slot2 = day_courses[i + 1]
                
                if course1.course.campus != course2.course.campus:
                    time_gap = slot2.start_section - slot1.end_section
                    if time_gap < self.config.min_campus_transfer_time:
                        conflicts.append(ConflictInfo(
                            conflict_type="campus",
                            course1=course1.course,
                            course2=course2.course,
                            description=f"校区转移时间不足：{course1.course.name}({course1.course.campus}) 到 {course2.course.name}({course2.course.campus}) 间隔{time_gap}节次",
                            severity="medium"
                        ))
    
    return conflicts
```

```python
class ScheduleEvaluator:
    """排课方案评估器"""
    
    def __init__(self, config: SchedulingConfig):
        self.config = config
    
    def evaluate_schedule(self, selected_courses: List[SelectedCourse],
                         credit_manager=None, config=None) -> ScheduleScore:
        """评估排课方案（修正的评分机制）"""
        score = ScheduleScore()

        if not selected_courses:
            return score

        # 关键修正：评分逻辑应该是先检查硬约束，满足了给100分，然后扣分
        # 这里假设传入的selected_courses已经满足硬约束（由排课引擎保证）
        # 如果不满足硬约束，分数为0
        score.hard_constraints_satisfied = True  # 由调用者设置

        if not score.hard_constraints_satisfied:
            score.total_score = 0.0
            score.credit_efficiency_score = 0.0
            return score

        # 满足硬约束的情况下，基于时间质量评分
        time_quality_score = self._calculate_time_quality_score(selected_courses)

        # 🔧 修正：增加学分匹配度评分的权重，确保学分匹配度是主要评分因素
        credit_match_score = self._calculate_credit_match_score(selected_courses, credit_manager)

        # 计算学分相关的惩罚（权重适中）
        overflow_penalty = self._calculate_credit_overflow_penalty(selected_courses, credit_manager) * 0.5
        gap_penalty = 0.0
        if config and config.credit_constraint_mode.value == "optimal":
            gap_penalty = self._calculate_credit_gap_penalty(selected_courses, credit_manager, config) * 0.5

        # 🔧 修正：最终分数计算 - 学分匹配度权重更高
        # 学分匹配度占70%，时间质量占30%
        score.total_score = max(0, credit_match_score * 0.7 + time_quality_score * 0.3 - overflow_penalty - gap_penalty)
        score.credit_efficiency_score = credit_match_score

        return score
```

**学分匹配度计算：**
```python
def _calculate_credit_match_score(self, selected_courses: List[SelectedCourse], 
                                 credit_manager=None) -> float:
    """计算学分匹配度分数（时间效率优化版本）"""
    if not credit_manager:
        return 50.0  # 默认分数

    # 按类别统计学分和完成周次
    category_credits = defaultdict(float)
    category_completion_weeks = defaultdict(lambda: 20)  # 默认第20周完成

    for course in selected_courses:
        category = course.custom_category
        category_credits[category] += course.course.credits
        
        # 计算课程的最早完成周次（基于时间段）
        if course.time_slots:
            min_week = min(min(slot.weeks) for slot in course.time_slots if slot.weeks)
            category_completion_weeks[category] = min(category_completion_weeks[category], min_week)

    # 计算时间效率分数
    total_efficiency = 0.0
    total_weight = 0.0

    for category, requirement in credit_manager.requirements.items():
        actual_credits = category_credits.get(category, 0.0)
        required_credits = requirement.required_credits
        completion_week = category_completion_weeks.get(category, 20)

        if actual_credits >= required_credits:
            # 已满足要求，计算时间效率
            # 越早完成分数越高：(21 - completion_week) / 20 * 100
            efficiency = (21 - completion_week) / 20 * 100
        else:
            # 未满足要求，根据完成度给分
            completion_ratio = actual_credits / required_credits if required_credits > 0 else 0
            efficiency = completion_ratio * 50  # 最高50分

        weight = required_credits  # 以要求学分作为权重
        total_efficiency += efficiency * weight
        total_weight += weight

    return (total_efficiency / total_weight) if total_weight > 0 else 0.0
```

```python
        """初始化后处理"""
        if not self.selected_courses:
            self.selected_courses = []
        
        # 计算基本统计信息
        self._calculate_statistics()
```

## 🔄 算法执行流程

### 1. 三阶段排课流程
```
输入课程列表 → 阶段1(线上课程处理) → 阶段2(约束满足排课) → 阶段3(结果合并) → 输出排课方案
```

### 2. 约束检查优先级
1. **硬约束**：时间冲突、校区冲突、学分要求（REQUIRED模式）
2. **软约束**：时间偏好、校区一致性、课程分布

### 3. 评分机制
- **学分匹配度**：70%权重，基于时间效率计算
- **时间质量**：30%权重，考虑课程分布和时间偏好
- **惩罚机制**：学分溢出惩罚、学分缺口惩罚

### 4. 算法优化策略
- **智能预筛选**：课程数量>15时进行预筛选
- **回溯算法**：生成满足约束的多个方案
- **贪心策略**：线上课程和简化排课使用贪心算法
- **诊断机制**：排课失败时提供详细诊断信息
