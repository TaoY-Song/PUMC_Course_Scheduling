```python
@dataclass
class TimeSlot:
    """时间段数据类"""
    weekday: int                 # 星期几 (1-7, 1=周一)
    start_section: int           # 开始节次
    end_section: int             # 结束节次
    weeks: List[int]             # 上课周次列表
    location: str = ""           # 上课地点
```

**实际方法实现：**
```python
def overlaps_with(self, other: 'TimeSlot') -> bool:
    """检查与另一个时间段是否重叠"""
    # 不同星期不重叠
    if self.weekday != other.weekday:
        return False
    
    # 检查节次是否重叠
    if (self.end_section < other.start_section or 
        other.end_section < self.start_section):
        return False
    
    # 检查周次是否重叠
    self_weeks_set = set(self.weeks)
    other_weeks_set = set(other.weeks)
    return bool(self_weeks_set & other_weeks_set)

def get_time_range_str(self) -> str:
    """获取时间范围字符串"""
    weekdays = ["", "周一", "周二", "周三", "周四", "周五", "周六", "周日"]
    weekday_str = weekdays[self.weekday] if 1 <= self.weekday <= 7 else f"第{self.weekday}天"
    
    if self.start_section == self.end_section:
        return f"{weekday_str} 第{self.start_section}节"
    else:
        return f"{weekday_str} 第{self.start_section}-{self.end_section}节"

def get_weeks_str(self) -> str:
    """获取周次字符串"""
    if not self.weeks:
        return ""
    
    # 对周次进行排序
    sorted_weeks = sorted(self.weeks)
    
    # 简化连续周次的表示
    if len(sorted_weeks) <= 3:
        return f"第{','.join(map(str, sorted_weeks))}周"
    
    # 检查是否为连续周次
    if sorted_weeks == list(range(sorted_weeks[0], sorted_weeks[-1] + 1)):
        return f"第{sorted_weeks[0]}-{sorted_weeks[-1]}周"
    else:
        return f"第{sorted_weeks[0]},{sorted_weeks[1]}...{sorted_weeks[-1]}周"
```

```python
def __post_init__(self):
    """初始化后处理 - 自动分配类别"""
    if not self.custom_category:
        self.custom_category = self._auto_assign_category()

def _auto_assign_category(self) -> str:
    """自动分配类别"""
    # 优先使用课程的自定义类别
    if self.course.custom_category and self.course.custom_category.strip():
        return self.course.custom_category.strip()
    
    # 否则使用课程的原始类别
    return self.course.category

def get_display_category(self) -> str:
    """获取显示用的类别"""
    return self.custom_category if self.custom_category else self.course.category

def has_time_conflict_with(self, other: 'SelectedCourse') -> bool:
    """检查与另一门课程是否有时间冲突"""
    for my_slot in self.time_slots:
        for other_slot in other.time_slots:
            if my_slot.overlaps_with(other_slot):
                return True
    return False
```

```python
def load_from_excel(self, file_path: str) -> bool:
    """从Excel文件加载课程数据"""
    try:
        print(f"正在加载课程数据: {file_path}")
        
        # 检查文件是否存在
        if not Path(file_path).exists():
            print(f"❌ 文件不存在: {file_path}")
            return False
        
        # 读取Excel文件
        df = pd.read_excel(file_path)
        print(f"✓ 成功读取Excel文件，共 {len(df)} 条记录")
        
        # 验证必要列
        if not self._validate_columns(df):
            return False
        
        # 清洗数据
        df = self._clean_data(df)
        
        # 转换为Course对象
        self.courses = []
        failed_count = 0
        
        for index, row in df.iterrows():
            try:
                course = self._create_course_from_row(row)
                if course:
                    self.courses.append(course)
            except Exception as e:
                failed_count += 1
                print(f"⚠️ 第 {index + 1} 行数据转换失败: {e}")
        
        # 生成加载报告
        self._generate_load_report(len(df), failed_count)
        
        print(f"✅ 数据加载完成: 成功加载 {len(self.courses)} 门课程")
        if failed_count > 0:
            print(f"⚠️ {failed_count} 条记录加载失败")
        
        return True
        
    except Exception as e:
        print(f"❌ 加载课程数据失败: {e}")
        return False
```

#### 2. _validate_columns() - 列验证
```python
def _validate_columns(self, df: pd.DataFrame) -> bool:
    """验证必要列是否存在"""
    required_columns = [
        "课程编码", "课程名称", "开课院系", "课程类别", "班次",
        "校区", "任课教师", "学分", "学时"
    ]

    missing_columns = [col for col in required_columns if col not in df.columns]
    if missing_columns:
        print(f"❌ Excel文件缺少必要列: {missing_columns}")
        print("请确保Excel文件包含以下列:")
        for col in required_columns:
            print(f"  - {col}")
        return False

    print("✓ Excel文件列验证通过")
    return True
```

#### 3. _clean_data() - 数据清洗
```python
def _clean_data(self, df: pd.DataFrame) -> pd.DataFrame:
    """清洗数据"""
    df = df.copy()

    # 去除字符串列的前后空格
    string_columns = ["课程编码", "课程名称", "开课院系", "课程类别",
                     "校区", "任课教师"]
    for col in string_columns:
        if col in df.columns:
            df[col] = df[col].astype(str).str.strip()

    # 处理数值列
    df["班次"] = pd.to_numeric(df["班次"], errors="coerce").fillna(1).astype(int)
    df["学分"] = pd.to_numeric(df["学分"], errors="coerce").fillna(0.0).astype(float)
    df["学时"] = pd.to_numeric(df["学时"], errors="coerce").fillna(0.0).astype(float)

    # 处理选课说明列（如果存在）
    if "选课说明" in df.columns:
        df["选课说明"] = df["选课说明"].fillna("").astype(str)

    # 过滤无效数据
    df = df.dropna(subset=["课程编码", "课程名称"])
    df = df[df["课程编码"].str.strip() != ""]
    df = df[df["课程名称"].str.strip() != ""]

    return df
```

#### 4. _create_course_from_row() - 对象创建
```python
def _create_course_from_row(self, row) -> Optional[Course]:
    """从数据行创建Course对象"""
    try:
        # 处理线上状态信息
        is_online_str = str(row.get("是否线上", "否")).strip()
        is_online = is_online_str == "是"

        # 处理自定义类别信息
        custom_category = str(row.get("自定义类别", "")).strip()

        return Course(
            code=str(row["课程编码"]).strip(),
            name=str(row["课程名称"]).strip(),
            department=str(row["开课院系"]).strip(),
            category=str(row["课程类别"]).strip(),
            class_num=int(row["班次"]),
            campus=str(row["校区"]).strip(),
            teacher=str(row["任课教师"]).strip(),
            credits=float(row["学分"]),
            hours=float(row["学时"]),
            description=str(row.get("选课说明", "")).strip(),
            is_online=is_online,
            custom_category=custom_category
        )
    except Exception as e:
        print(f"创建课程对象失败: {e}")
        return None
```

```python
def _generate_load_report(self, total_records: int, failed_count: int):
    """生成加载报告"""
    self.load_report = {
        "total_records": total_records,
        "successful_records": len(self.courses),
        "failed_records": failed_count,
        "success_rate": f"{(len(self.courses) / total_records * 100):.1f}%" if total_records > 0 else "0%",
        "categories": list(set(course.category for course in self.courses)),
        "departments": list(set(course.department for course in self.courses)),
        "campuses": list(set(course.campus for course in self.courses))
    }
```

## 🔄 数据处理流程

### 1. 课程数据加载流程
```
Excel文件 → 文件存在性检查 → 列验证 → 数据清洗 → 对象转换 → 报告生成
```

### 2. 数据验证规则
- **必需列验证**：9个核心列必须存在
- **数据类型转换**：自动处理数值类型转换和错误值
- **空值处理**：字符串列去除空格，数值列填充默认值
- **无效数据过滤**：移除课程编码或名称为空的记录

### 3. 导入导出数据格式
- **导出格式**：三工作表（课程列表、周课表、统计信息）
- **时间段拆分**：最多支持5个时间段的详细信息
- **动态列名**：自动复制原始Excel文件的列结构
- **学分统计**：可选的学分完成情况统计表

### 4. 错误处理机制
- **逐行处理**：单行错误不影响其他数据处理
- **详细日志**：记录每个处理步骤的成功/失败信息
- **统计报告**：提供完整的处理结果统计
- **异常恢复**：关键错误时返回部分成功结果
